FONT_FAMILIES = (
    "Microsoft JhengHei UI",
    "Microsoft JhengHei",
    "Noto Sans TC",
    "PingFang TC",
    "Segoe UI",
)


def make_font(family: str = "", pixel_size: int = 12) -> QFont:
    font = QFont()
    if family:
        font.setFamily(family)
    else:
        available = set(QFontDatabase.families())
        for candidate in FONT_FAMILIES:
            if candidate in available:
                font.setFamily(candidate)
                break
    font.setPixelSize(pixel_size)
    return font

"""不取得焦點、可穿透滑鼠的字幕視窗。"""

from __future__ import annotations

import copy

from PySide6.QtCore import (
    QAbstractAnimation,
    QEasingCurve,
    QPointF,
    QPropertyAnimation,
    QRect,
    QRectF,
    Qt,
)
from PySide6.QtGui import (
    QColor,
    QFontMetrics,
    QGuiApplication,
    QPaintEvent,
    QPainter,
    QPainterPath,
    QPen,
    QScreen,
    QTextCharFormat,

    QTextOption,
)
from PySide6.QtWidgets import QWidget

from livesub.config import DisplaySettings
from livesub.core.captions import BoardSnapshot, EMPTY_SNAPSHOT

from .theme import (
    OVERLAY_BLOCK_GAP,
    OVERLAY_PADDING,
    OVERLAY_RADIUS,
    make_font,
)


class OverlayWindow(QWidget):
    def __init__(self, settings: DisplaySettings):
        flags = (
            Qt.WindowType.FramelessWindowHint
            | Qt.WindowType.WindowStaysOnTopHint
            | Qt.WindowType.Tool
            | Qt.WindowType.WindowTransparentForInput
            | Qt.WindowType.WindowDoesNotAcceptFocus
        )
        super().__init__(None, flags)
        self.setAttribute(Qt.WidgetAttribute.WA_TranslucentBackground)
        self.setAttribute(Qt.WidgetAttribute.WA_ShowWithoutActivating)
        self.setAttribute(Qt.WidgetAttribute.WA_TransparentForMouseEvents)

        self._settings = copy.deepcopy(settings)
        self._snapshot = EMPTY_SNAPSHOT
        self._block_count = 1
        self._screen: QScreen | None = None
        self._fallback_to_primary = False
        self._paths: list[tuple[QPainterPath, bool]] = []
        self._font = make_font(settings.font_family, settings.font_size)

        self.fade_animation = QPropertyAnimation(self, b"windowOpacity", self)
        self.fade_animation.setDuration(300)
        self.fade_animation.setEasingCurve(QEasingCurve.Type.InOutQuad)
        self._opacity_target = 0.0
        self.setWindowOpacity(0.0)

        app = QGuiApplication.instance()
        if app is None:
            raise RuntimeError("建立字幕視窗前必須先建立 QApplication")
        app.screenAdded.connect(self._screens_changed)
        app.screenRemoved.connect(self._screen_removed)
        app.primaryScreenChanged.connect(self._screens_changed)
        self.apply_settings(settings)

    def apply_settings(self, settings: DisplaySettings) -> None:
        if settings.screen != self._settings.screen:
            self._fallback_to_primary = False
        self._settings = copy.deepcopy(settings)
        self._font = make_font(settings.font_family, settings.font_size)
        self.setFont(self._font)
        self._refresh()

    def set_block_count(self, count: int) -> None:
        """收到新模式的快照之前，可先預留對應高度。"""
        count = max(1, int(count))
        if count != self._block_count:
            self._block_count = count
            self._refresh()

    def _choose_screen(self) -> QScreen | None:
        screens = QGuiApplication.screens()
        index = self._settings.screen
        if not self._fallback_to_primary and 0 <= index < len(screens):
            return screens[index]
        return QGuiApplication.primaryScreen()

    def _bind_screen(self) -> None:
        screen = self._choose_screen()
        if screen is self._screen:
            return
        if self._screen is not None:
            try:
                self._screen.geometryChanged.disconnect(self._geometry_changed)
            except (RuntimeError, TypeError):
                pass
        self._screen = screen
        if screen is not None:
            screen.geometryChanged.connect(self._geometry_changed)

    def _screens_changed(self, *_: object) -> None:
        self._refresh()

    def _screen_removed(self, screen: QScreen) -> None:
        if screen is self._screen:
            self._fallback_to_primary = True
        self._refresh()

    def _geometry_changed(self, _: QRect) -> None:
        self._refresh()

    def target_geometry(self) -> QRect:
        screen = self._choose_screen()
        geo = screen.geometry() if screen is not None else QRect(0, 0, 800, 600)
        d = self._settings
        width = max(1, round(geo.width() * d.width_pct / 100))
        line_height = QFontMetrics(self._font).lineSpacing()
        height = (
            self._block_count * d.max_lines * line_height
            + (self._block_count - 1) * OVERLAY_BLOCK_GAP
            + 2 * OVERLAY_PADDING
        )
        x = geo.left() + (geo.width() - width) // 2
        if d.position == "top":
            y = geo.top() + d.margin_px
        else:
            # 依介面契約使用 QRect.bottom() 的含尾端座標。
            y = geo.bottom() - d.margin_px - height
        return QRect(x, y, width, max(1, height))

    def _refresh(self) -> None:
        self._bind_screen()
        geometry = self.target_geometry()
        if self.geometry() != geometry:
            self.setGeometry(geometry)
        self._cache_paths()
        self.update()

    def show_snapshot(self, snap: BoardSnapshot) -> None:
        self._snapshot = snap
        if snap.blocks:
            self._block_count = len(snap.blocks)
        self._refresh()
        self._animate_opacity(1.0 if snap.visible and not snap.empty else 0.0)

    def _animate_opacity(self, target: float) -> None:
        if target == self._opacity_target:
            return
        self._opacity_target = target
        self.fade_animation.stop()
        current = self.windowOpacity()
        if abs(current - target) < 0.001:
            self.setWindowOpacity(target)
            return
        self.fade_animation.setStartValue(current)
        self.fade_animation.setEndValue(target)
        self.fade_animation.start()

    @staticmethod
    def _cursor_x(line: object, position: int) -> float:
        result = line.cursorToX(position)
        # PySide6 版本可能將輸入／輸出參數包成 tuple。
        return float(result[0] if isinstance(result, tuple) else result)

    def _cache_paths(self) -> None:
        self._paths.clear()
        if self._snapshot.empty:
            return

        d = self._settings
        metrics = QFontMetrics(self._font)
        line_height = metrics.lineSpacing()
        block_height = d.max_lines * line_height
        available_width = max(1, self.width() - 2 * OVERLAY_PADDING)

        for block_index, block in enumerate(self._snapshot.blocks):
            if block_index >= self._block_count:
                break

            # Qt 的文字索引以 UTF-16 code unit 計算，不是 Python 字元數。
            text = block.text.replace("\r", " ").replace("\n", "\u2028")
            encoded = text.encode("utf-16-le")
            spans: list[tuple[int, int, bool]] = []
            formats: list[QTextLayout.FormatRange] = []
            offset = 0
            for span in block.spans:
                length = len(span.text.encode("utf-16-le")) // 2
                spans.append((offset, offset + length, span.dim))
                fmt = QTextCharFormat()
                fmt.setForeground(QColor(255, 255, 255, 153 if span.dim else 255))
                item = QTextLayout.FormatRange()
                item.start = offset
                item.length = length
                item.format = fmt
                formats.append(item)
                offset += length

            layout = QTextLayout(text, self._font)
            option = QTextOption()
            option.setWrapMode(QTextOption.WrapMode.WrapAtWordBoundaryOrAnywhere)
            option.setAlignment(Qt.AlignmentFlag.AlignLeft)
            option.setTextDirection(Qt.LayoutDirection.LeftToRight)
            layout.setTextOption(option)
            layout.setFormats(formats)

            lines = []
            layout.beginLayout()
            while True:
                line = layout.createLine()
                if not line.isValid():
                    break
                line.setLineWidth(available_width)
                line.setPosition(QPointF(0, len(lines) * line_height))
                lines.append(line)
            layout.endLayout()

            kept = lines[-d.max_lines:]
            block_top = OVERLAY_PADDING + block_index * (
                block_height + OVERLAY_BLOCK_GAP
            )
            for row, line in enumerate(kept):
                start = line.textStart()
                end = start + line.textLength()
                left = OVERLAY_PADDING + (
                    available_width - line.naturalTextWidth()
                ) / 2
                baseline = block_top + row * line_height + metrics.ascent()

                for span_start, span_end, dim in spans:
                    a, b = max(start, span_start), min(end, span_end)
                    if a >= b:
                        continue
                    fragment = encoded[a * 2:b * 2].decode("utf-16-le")
                    fragment = fragment.replace("\u2028", "")
                    if not fragment:
                        continue
                    path = QPainterPath()
                    path.addText(
                        QPointF(left + self._cursor_x(line, a), baseline),
                        self._font,
                        fragment,
                    )
                    self._paths.append((path, dim))

    def paintEvent(self, event: QPaintEvent) -> None:
        if self._snapshot.empty:
            return

        painter = QPainter(self)
        painter.setRenderHint(QPainter.RenderHint.Antialiasing)
        d = self._settings

        if d.style == "box":
            painter.setPen(Qt.PenStyle.NoPen)
            painter.setBrush(
                QColor(0, 0, 0, round(255 * max(0.0, min(1.0, d.box_opacity))))
            )
            painter.drawRoundedRect(
                QRectF(self.rect()), OVERLAY_RADIUS, OVERLAY_RADIUS
            )

        stroke_width = (
            max(1.0, d.font_size / 9)
            if d.style == "outline"
            else max(1.0, d.font_size / 24)
        )
        pen = QPen(QColor(0, 0, 0, 235))
        pen.setWidthF(stroke_width)
        pen.setJoinStyle(Qt.PenJoinStyle.RoundJoin)

        for path, dim in self._paths:
            painter.save()
            painter.setOpacity(0.6 if dim else 1.0)
            if d.style == "outline":
                painter.save()
                painter.translate(1.5, 2.0)
                painter.fillPath(path, QColor(0, 0, 0, 105))
                painter.restore()
            painter.strokePath(path, pen)
            painter.fillPath(path, QColor("white"))
            painter.restore()
        painter.end()

    @staticmethod
    def _combo(items: list[tuple[str, object]], current: object) -> QComboBox:
        combo = QComboBox()
        with QSignalBlocker(combo):
            for label, value in items:
                combo.addItem(label, value)
            index = combo.findData(current)
            combo.setCurrentIndex(max(0, index))
        return combo

    @staticmethod
    def _slider(
        minimum: int, maximum: int, value: int, suffix: str
    ) -> tuple[QSlider, QLabel, QWidget]:
        row = QWidget()
        layout = QHBoxLayout(row)
        layout.setContentsMargins(0, 0, 0, 0)
        layout.setSpacing(6)
        slider = QSlider(Qt.Orientation.Horizontal)
        slider.setRange(minimum, maximum)
        with QSignalBlocker(slider):
            slider.setValue(value)
        label = QLabel(f"{slider.value()}{suffix}")
        label.setMinimumWidth(43)
        label.setAlignment(Qt.AlignmentFlag.AlignRight | Qt.AlignmentFlag.AlignVCenter)
        slider.valueChanged.connect(
            lambda number: label.setText(f"{number}{suffix}")
        )
        layout.addWidget(slider, 1)
        layout.addWidget(label)
        return slider, label, row

    def _combo_changed(
        self, control: QComboBox, section: str, field: str
    ) -> None:
        if control.currentIndex() >= 0:
            self._change(section, field, control.currentData())

    def _change(self, section: str, field: str, value: object) -> None:
        target = getattr(self._settings, section)
        if getattr(target, field) == value:
            return
        setattr(target, field, value)
        if section == "display" and field == "style":
            self._update_opacity_enabled()
        self.settingsChanged.emit(copy.deepcopy(self._settings))

    def _update_opacity_enabled(self) -> None:
        enabled = self._settings.display.style == "box"
        self.opacity_slider.setEnabled(enabled)
        self.opacity_value.setEnabled(enabled)

    def _set_advanced(self, expanded: bool) -> None:
        self.advanced_button.setArrowType(
            Qt.ArrowType.DownArrow if expanded else Qt.ArrowType.RightArrow
        )
        self.advanced_group.setVisible(expanded)
        self.layout().activate()
        self.resize(self.width(), self.sizeHint().height())

    def _refresh_screens(self, *_: object) -> None:
        for screen in self._watched_screens:
            try:
                screen.geometryChanged.disconnect(self._refresh_screens)
            except (RuntimeError, TypeError):
                pass
        self._watched_screens = list(QGuiApplication.screens())
        with QSignalBlocker(self.screen_combo):
            self.screen_combo.clear()
            for index, screen in enumerate(self._watched_screens):
                geo = screen.geometry()
                name = screen.name() or f"螢幕 {index + 1}"
                self.screen_combo.addItem(
                    f"{name}（{geo.width()} × {geo.height()}）", index
                )
            selected = self.screen_combo.findData(self._settings.display.screen)
            if selected < 0:
                primary = QGuiApplication.primaryScreen()
                selected = next(
                    (i for i, screen in enumerate(self._watched_screens)
                     if screen is primary),
                    -1,
                )
            self.screen_combo.setCurrentIndex(selected)
        for screen in self._watched_screens:
            screen.geometryChanged.connect(self._refresh_screens)

    def set_devices(self, devices: list[str], current: str) -> None:
        names = list(dict.fromkeys(name for name in devices if name))
        if current and current not in names:
            names.append(current)
        with QSignalBlocker(self.device_combo):
            self.device_combo.clear()
            self.device_combo.addItem("系統預設", "")
            for name in names:
                self.device_combo.addItem(name, name)
            self.device_combo.setCurrentIndex(self.device_combo.findData(current))
        self._settings.audio.device = current

    def set_status(self, status: PipelineStatus) -> None:
        self._status = status
        busy = status.state in (PipelineState.LOADING, PipelineState.STOPPING)
        with QSignalBlocker(self.start_button):
            self.start_button.setText(
                "停止" if status.running or status.state == PipelineState.STOPPING
                else "開始"
            )
            self.start_button.setEnabled(not busy)

        parts = [STATE_LABELS[status.state]]
        if status.profile_label:
            parts.append(status.profile_label)
        if status.latency_s is not None and math.isfinite(status.latency_s):
            parts.append(f"延遲 {max(0.0, status.latency_s):.1f} 秒")
        if status.segmented:
            parts.append("分段字幕")
        self.status_label.setText(" · ".join(parts))
        self.warning_label.setText(status.message)
        self.warning_label.setVisible(bool(status.message))
        level = status.level if math.isfinite(status.level) else 0.0
        with QSignalBlocker(self.level_bar):
            self.level_bar.setValue(round(max(0.0, min(1.0, level)) * 100))

    def set_hotkey_ok(self, ok: bool) -> None:
        self.hotkey_label.setText(
            "Ctrl+Alt+H 顯示/隱藏面板 · Ctrl+Alt+S 開始/停止"
            if ok else "全域熱鍵註冊失敗，請用系統匣叫回面板"
        )

    def _start_stop(self) -> None:
        if self._status.state in (PipelineState.LOADING, PipelineState.STOPPING):
            return
        if self._status.running:
            self.stopRequested.emit()
        else:
            self.startRequested.emit()

    def _hide_panel(self) -> None:
        if self.can_hide:
            self.hide()
            self.hideRequested.emit()

    def closeEvent(self, event: QCloseEvent) -> None:
        if self.can_hide:
            event.ignore()
            self._hide_panel()
        else:
            event.accept()
            self.quitRequested.emit()

from __future__ import annotations

from PySide6.QtCore import QObject, QRectF, Qt, Signal
from PySide6.QtGui import QColor, QIcon, QPainter, QPen, QPixmap
from PySide6.QtWidgets import QMenu, QSystemTrayIcon

from .theme import PANEL_QSS


class TrayIcon(QSystemTrayIcon):
    toggleRequested = Signal()
    startStopRequested = Signal()
    quitRequested = Signal()

    def __init__(self, parent: QObject | None = None):
        super().__init__(parent)
        self.setIcon(self._make_icon())
        self.setToolTip("即時字幕")

        self._menu = QMenu()
        self._menu.setStyleSheet(PANEL_QSS)
        toggle_action = self._menu.addAction("顯示/隱藏面板")
        toggle_action.triggered.connect(self.toggleRequested.emit)
        self._start_action = self._menu.addAction("開始")
        self._start_action.triggered.connect(self.startStopRequested.emit)
        self._menu.addSeparator()
        quit_action = self._menu.addAction("結束")
        quit_action.triggered.connect(self.quitRequested.emit)

        self.setContextMenu(self._menu)
        self.activated.connect(self._activated)
        self.destroyed.connect(self._menu.deleteLater)

    @staticmethod
    def _make_icon() -> QIcon:
        icon = QIcon()
        for size in (16, 24, 32, 48, 64):
            pixmap = QPixmap(size, size)
            pixmap.fill(Qt.GlobalColor.transparent)
            painter = QPainter(pixmap)
            painter.setRenderHint(QPainter.RenderHint.Antialiasing)
            painter.scale(size / 32, size / 32)
            painter.setPen(Qt.PenStyle.NoPen)
            painter.setBrush(QColor("#306da8"))
            painter.drawRoundedRect(QRectF(1, 3, 30, 26), 6, 6)

            pen = QPen(QColor("white"))
            pen.setWidthF(2.5)
            pen.setCapStyle(Qt.PenCapStyle.RoundCap)
            painter.setPen(pen)
            painter.drawLine(7, 12, 25, 12)
            painter.drawLine(7, 20, 15, 20)
            painter.drawLine(20, 20, 25, 20)
            painter.end()
            icon.addPixmap(pixmap)
        return icon

    def set_running(self, running: bool) -> None:
        self._start_action.setText("停止" if running else "開始")

    def _activated(self, reason: QSystemTrayIcon.ActivationReason) -> None:
        if reason == QSystemTrayIcon.ActivationReason.Trigger:
            self.toggleRequested.emit()

給 Claude 的整合說明：

- `settingsChanged` 連到設定儲存與 `overlay.apply_settings(settings.display)`；模式改變後請重新送出 snapshot，也可先呼叫 `set_block_count(2 if mode == "bilingual" else 1)`。
- 連接面板的 `startRequested`、`stopRequested`、`hideRequested`、`quitRequested`，以及系統匣的三個 Signal。建立後由整合層呼叫 `overlay.show()`、`panel.show()`、`tray.show()`。
- Pipeline 的 snapshot/status 請透過 Qt queued signal 傳到 GUI 執行緒；設定儲存的 0.5 秒防抖由整合層負責。
- 依指定高度公式，各區塊使用相同字級，未套用 `secondary_scale`。複雜連寫文字跨 Span 的 shaping 不在目前中英文範圍內。
- 環境政策封鎖執行工具，以上測試尚未實跑；請註冊 `gui` marker，再以 `QT_QPA_PLATFORM=offscreen` 執行這兩個測試檔。實際滑鼠穿透、置頂與淡出效果仍需桌面環境驗證。
